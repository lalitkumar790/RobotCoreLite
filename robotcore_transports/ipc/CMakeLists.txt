cmake_minimum_required(VERSION 3.16)
project(robotcore_ipc LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(robotcore_ipc STATIC src/ipc.cpp)
target_include_directories(robotcore_ipc PUBLIC include ../../robotcore_core/include)
target_compile_features(robotcore_ipc PUBLIC cxx_std_17)
target_link_libraries(robotcore_ipc PUBLIC Threads::Threads)
target_compile_options(robotcore_ipc PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
 add_executable(ipc_test tests/ipc_test.cpp)
 target_link_libraries(ipc_test PRIVATE robotcore_ipc)
 target_compile_options(ipc_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 add_test(NAME ipc_contract COMMAND ipc_test)
 set_tests_properties(ipc_contract PROPERTIES TIMEOUT 20)

 target_compile_options(ipc_stress_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 add_test(NAME ipc_concurrent_publish COMMAND ipc_stress_test)
 set_tests_properties(ipc_concurrent_publish PROPERTIES TIMEOUT 20)
 add_executable(ipc_reconnect_test tests/ipc_reconnect_test.cpp)
 target_link_libraries(ipc_reconnect_test PRIVATE robotcore_ipc)
 target_compile_options(ipc_reconnect_test PRIVATE -Wall -Wextra -Wpedantic -Werror)

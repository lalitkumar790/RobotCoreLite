cmake_minimum_required(VERSION 3.16)
project(robotcore_shared_memory LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(robotcore_shared_memory STATIC src/shared_memory.cpp)
target_include_directories(robotcore_shared_memory PUBLIC include ../../robotcore_core/include)
target_compile_features(robotcore_shared_memory PUBLIC cxx_std_17)
target_link_libraries(robotcore_shared_memory PUBLIC Threads::Threads)
target_compile_options(robotcore_shared_memory PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(shared_memory_test tests/shared_memory_test.cpp)
  target_link_libraries(shared_memory_test PRIVATE robotcore_shared_memory)
  target_compile_options(shared_memory_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME shared_memory_cross_process COMMAND shared_memory_test)
  set_tests_properties(shared_memory_cross_process PROPERTIES TIMEOUT 20)
  add_executable(shared_memory_edges tests/shared_memory_edges.cpp)
  target_link_libraries(shared_memory_edges PRIVATE robotcore_shared_memory)
  target_compile_options(shared_memory_edges PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME shared_memory_owner_and_crash COMMAND shared_memory_edges)
  set_tests_properties(shared_memory_owner_and_crash PROPERTIES TIMEOUT 20)
  add_executable(shared_memory_liveness tests/shared_memory_liveness.cpp)
  target_link_libraries(shared_memory_liveness PRIVATE robotcore_shared_memory)
  target_compile_options(shared_memory_liveness PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME shared_memory_liveness_and_fail_closed COMMAND shared_memory_liveness)
  set_tests_properties(shared_memory_liveness_and_fail_closed PROPERTIES TIMEOUT 20)
  add_executable(shared_memory_stale_session tests/shared_memory_stale_session.cpp)
  target_link_libraries(shared_memory_stale_session PRIVATE robotcore_shared_memory)
  target_compile_options(shared_memory_stale_session PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME shared_memory_stale_session_reject COMMAND shared_memory_stale_session)
  set_tests_properties(shared_memory_stale_session_reject PROPERTIES TIMEOUT 20)
  add_executable(shared_memory_borrowed tests/shared_memory_borrowed.cpp)
  target_link_libraries(shared_memory_borrowed PRIVATE robotcore_shared_memory)
  target_compile_options(shared_memory_borrowed PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME shared_memory_borrowed_cross_process COMMAND shared_memory_borrowed)
  set_tests_properties(shared_memory_borrowed_cross_process PROPERTIES TIMEOUT 30)
  add_executable(shared_memory_publish_stop_race ../../tests/native_shared_memory_publish_stop.cpp)
  target_link_libraries(shared_memory_publish_stop_race PRIVATE robotcore_shared_memory)
  target_compile_options(shared_memory_publish_stop_race PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME shared_memory_publish_stop_lifetime COMMAND shared_memory_publish_stop_race)
  set_tests_properties(shared_memory_publish_stop_lifetime PROPERTIES TIMEOUT 60)
endif()

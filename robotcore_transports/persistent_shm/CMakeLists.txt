cmake_minimum_required(VERSION 3.16)
project(robotcore_persistent_shm LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(robotcore_persistent_shm STATIC src/persistent_shm.cpp)
target_include_directories(robotcore_persistent_shm PUBLIC include ../../robotcore_core/include)
target_compile_features(robotcore_persistent_shm PUBLIC cxx_std_17)
target_link_libraries(robotcore_persistent_shm PUBLIC Threads::Threads)
target_compile_options(robotcore_persistent_shm PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(persistent_shm_test tests/persistent_shm_test.cpp)
  target_link_libraries(persistent_shm_test PRIVATE robotcore_persistent_shm)
  target_compile_options(persistent_shm_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME persistent_session_lifecycle COMMAND persistent_shm_test)
  set_tests_properties(persistent_session_lifecycle PROPERTIES TIMEOUT 30)
  add_executable(persistent_fault_test tests/persistent_fault_test.cpp)
  target_link_libraries(persistent_fault_test PRIVATE robotcore_persistent_shm)
  target_compile_options(persistent_fault_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME persistent_fault_injection COMMAND persistent_fault_test)
  set_tests_properties(persistent_fault_injection PROPERTIES TIMEOUT 30)
endif()

add_executable(transport_bench_persistent_shm ../../benchmarks/latency/transport_benchmark.cpp)
target_compile_definitions(transport_bench_persistent_shm PRIVATE ROBOTCORE_BENCH_PSHM)
target_link_libraries(transport_bench_persistent_shm PRIVATE robotcore_persistent_shm)
target_compile_options(transport_bench_persistent_shm PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(persistent_region_setup ../../benchmarks/setup/persistent_region_setup.cpp)
target_link_libraries(persistent_region_setup PRIVATE robotcore_persistent_shm)
target_compile_options(persistent_region_setup PRIVATE -Wall -Wextra -Wpedantic -Werror)

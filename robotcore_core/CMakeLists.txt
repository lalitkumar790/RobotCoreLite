cmake_minimum_required(VERSION 3.16)
project(robotcore_core LANGUAGES CXX)
add_library(robotcore_core INTERFACE)
target_include_directories(robotcore_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(robotcore_core INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  add_executable(robotcore_core_test tests/core_test.cpp)
  target_link_libraries(robotcore_core_test PRIVATE robotcore_core)
  target_compile_options(robotcore_core_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME robotcore_core_test COMMAND robotcore_core_test)
endif()
